# CLI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the `openapi2code` CLI binary — `openapi2code pull <source> --target ts (--output <dir> | --out-file <path>)` — wrapping the existing `pkg/engine` to read a spec from a file/URL/stdin and write generated TypeScript to disk.

**Architecture:** A thin `cmd/openapi2code/main.go` entrypoint calls `internal/cli.Run`, which is the only new package that performs file/network I/O. `internal/cli` depends on `pkg/engine`; `pkg/engine` is untouched and stays I/O-free.

**Tech Stack:** Go standard library only (`flag`, `net/http`, `os`, `io`, `path/filepath`) — no new third-party dependencies.

**Spec:** `docs/superpowers/specs/2026-09-10-cli-design.md`

## Global Constraints

- `internal/cli` is the only new package that performs file or network I/O; `pkg/engine` is not modified.
- No new third-party dependencies — stdlib only.
- `--target` defaults to `"ts"` if omitted; any other value returns an error containing "not yet implemented", not a generic flag error.
- Exactly one of `--output` (modular) or `--out-file` (monolithic) is required; both or neither is an error.
- `<source>`: `http://`/`https://` prefix → HTTP GET (30s timeout); exactly `-` → read from stdin; otherwise → local file path.
- Output writing overwrites only the exact files the engine generated; anything else already present at the destination is left untouched. Parent directories are created as needed.
- Every failure prints one line to stderr (`err.Error()`, no stack trace) and the process exits 1; success exits 0.

package cli

import (
	"flag"
	"fmt"
)

// Options is the fully validated, parsed form of the CLI's arguments.
type Options struct {
	Source  string
	Target  string
	OutDir  string
	OutFile string
}

// parseArgs parses and validates the "pull" subcommand's arguments. It
// does not touch the filesystem or network — see resolveSource/writeOutput.
func parseArgs(args []string) (Options, error) {
	if len(args) == 0 || args[0] != "pull" {
		return Options{}, fmt.Errorf("usage: openapi2code pull <source> --target ts (--output <dir> | --out-file <path>)")
	}

	fs := flag.NewFlagSet("pull", flag.ContinueOnError)
	target := fs.String("target", "ts", `generation target (only "ts" is currently supported)`)
	outDir := fs.String("output", "", "write modular output (one file per model + barrel index.ts) into this directory")
	outFile := fs.String("out-file", "", "write monolithic output to this single file")

	if err := fs.Parse(args[1:]); err != nil {
		return Options{}, err
	}

	rest := fs.Args()
	if len(rest) != 1 {
		return Options{}, fmt.Errorf("expected exactly one <source> argument, got %d", len(rest))
	}

	if *target != "ts" {
		return Options{}, fmt.Errorf("target %q not yet implemented (only \"ts\" is currently supported)", *target)
	}
	if (*outDir == "") == (*outFile == "") {
		return Options{}, fmt.Errorf("exactly one of --output or --out-file is required")
	}

	return Options{Source: rest[0], Target: *target, OutDir: *outDir, OutFile: *outFile}, nil
}
```

package cli

import (
	"fmt"
	"io"
	"net/http"
	"os"
	"strings"
	"time"
)

var httpClient = &http.Client{Timeout: 30 * time.Second}

// resolveSource reads the spec bytes named by source: an http(s):// URL is
// fetched, "-" reads from stdin, anything else is treated as a local file
// path.
func resolveSource(source string, stdin io.Reader) ([]byte, error) {
	switch {
	case strings.HasPrefix(source, "http://") || strings.HasPrefix(source, "https://"):
		return fetchURL(source)
	case source == "-":
		return io.ReadAll(stdin)
	default:
		return os.ReadFile(source)
	}
}

func fetchURL(url string) ([]byte, error) {
	resp, err := httpClient.Get(url)
	if err != nil {
		return nil, fmt.Errorf("fetch %s: %w", url, err)
	}
	defer resp.Body.Close()
	if resp.StatusCode < 200 || resp.StatusCode >= 300 {
		return nil, fmt.Errorf("fetch %s: unexpected status %s", url, resp.Status)
	}
	return io.ReadAll(resp.Body)
}
```

package cli

import (
	"os"
	"path/filepath"

	"github.com/tarikomercehajic/openapi2code/pkg/engine"
)

// writeOutput writes out to disk: to the single file outFile if set,
// otherwise as one file per out.Files entry inside outDir. Existing files
// at the exact destination paths are overwritten; nothing else at the
// destination is touched.
func writeOutput(out engine.Output, outDir, outFile string) error {
	if outFile != "" {
		if err := os.MkdirAll(filepath.Dir(outFile), 0755); err != nil {
			return err
		}
		return os.WriteFile(outFile, []byte(out.Files["index.ts"]), 0644)
	}

	if err := os.MkdirAll(outDir, 0755); err != nil {
		return err
	}
	for name, content := range out.Files {
		if err := os.WriteFile(filepath.Join(outDir, name), []byte(content), 0644); err != nil {
			return err
		}
	}
	return nil
}
```

package cli

import (
	"fmt"
	"io"

	"github.com/tarikomercehajic/openapi2code/pkg/engine"
)

// Run executes the CLI: parses args, resolves the source, generates code,
// and writes it out. It returns the process exit code (0 on success, 1 on
// any failure) rather than calling os.Exit itself, and takes stdin/stderr
// as parameters so it's testable without real process I/O.
func Run(args []string, stdin io.Reader, stderr io.Writer) int {
	opts, err := parseArgs(args)
	if err != nil {
		fmt.Fprintln(stderr, err)
		return 1
	}

	data, err := resolveSource(opts.Source, stdin)
	if err != nil {
		fmt.Fprintln(stderr, err)
		return 1
	}

	doc, err := engine.Parse(data)
	if err != nil {
		fmt.Fprintln(stderr, err)
		return 1
	}

	out, err := engine.GenerateTS(doc, engine.TSOptions{Modular: opts.OutDir != ""})
	if err != nil {
		fmt.Fprintln(stderr, err)
		return 1
	}

	if err := writeOutput(out, opts.OutDir, opts.OutFile); err != nil {
		fmt.Fprintln(stderr, err)
		return 1
	}

	return 0
}
```
